# =============================================================================
# MyGist MCP Server Dependencies
# =============================================================================
# Versions are pinned exactly: the app relies on Starlette middleware behaviour
# (contextvar/request.state propagation) and the fastmcp 2.x tool API, both of
# which can change across releases. Unpinned ranges previously let production
# rebuilds drift onto fastmcp 3.x / starlette 1.x. Bump deliberately + re-test.

# FastMCP - Modern MCP SDK with HTTP transport
fastmcp==2.14.2

# ASGI server
uvicorn[standard]==0.38.0

# Web framework (FastMCP dependency, explicit for middleware)
starlette==0.50.0

# Environment variables
python-dotenv==1.2.1

# Data validation (FastMCP dependency)
pydantic==2.12.5

# Legacy FastAPI for main.py (persona manager API)
fastapi==0.123.9

# Postgres driver
psycopg[binary]==3.3.4
psycopg-pool==3.3.1

# Password hashing (tokens stay sha256; passwords need a slow hash)
bcrypt==5.0.0

# Verifying Better Auth's JWTs against its JWKS endpoint (jwt_auth.py). Both
# arrived transitively via fastmcp before this; declared explicitly because an
# unrelated fastmcp bump dropping either would break authentication, and the
# failure would look like "everyone is logged out" rather than an import error.
# cryptography is not optional here -- PyJWT cannot verify Ed25519 without it,
# and Ed25519 is what Better Auth signs with by default.
PyJWT==2.10.1
cryptography==46.0.3

# Embedding providers
httpx==0.28.1

# Section pack manifest validation (boot-time)
jsonschema==4.25.1

# Schema migrations. Pulls in SQLAlchemy, which the application itself does
# not use -- there are no ORM models and every query is raw psycopg. Alembic
# is here for versioned, ordered, reviewable migrations; autogenerate is
# unavailable by design and revisions are hand-written SQL.
alembic==1.18.5
